\end{tresc}
\begin{wariant}{A} 10 km \end{wariant}
\begin{wariant}{B} 20 km \end{wariant}
\begin{wariant}{C} 30 km \end{wariant}
\begin{wariant}{D} 40 km \end{wariant}
\begin{wariant}{E} 50 km \end{wariant}
