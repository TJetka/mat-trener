\end{tresc}
\begin{wariant}{A} 1 \end{wariant}
\begin{wariant}{B} 2 \end{wariant}
\begin{wariant}{C} 3 \end{wariant}
\begin{wariant}{D} będzie ich tyle samo \end{wariant}
\begin{wariant}{E} czerwonych będzie więcej niż zielonych \end{wariant}
