\begin{tresc}

\end{tresc}

\begin{wariant}{B} 23 \end{wariant}
\begin{wariant}{C} 26 \end{wariant}
\begin{wariant}{D} 29 \end{wariant}
\begin{wariant}{E} 32 \end{wariant}
